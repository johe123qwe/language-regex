# 自动修复规则文件 - 格鲁吉亚语
# 使用说明:
# 1. 此文件放在 src/config/ 目录，命名格式：{语种代码}_fix.txt
# 2. # 开头为注释行，不生效
# 3. 格式为 Python dict：{r'正则匹配模式': '替换内容'}
# 4. 替换作用于 docx 表格外语列（最右侧列）的每个段落
# 5. 若无此文件，修复步骤自动跳过
# 6. 注释里的 @标签 需与 ka_txt.txt 中的检查描述完全一致，才能关联修复状态
# 7. 规则按书写顺序依次执行，改动清理类规则务必排在空格规则之前
#
{
#=== 隐形字符清理（须最先执行）===#
    r'[\u00ad\u200b-\u200f\u202a-\u202e\ufeff]': '',   # @不可见字符
    r'\u00a0': ' ',     # @不间断空格NBSP

#=== 空格修复 ===#
    r'^ +': '',         # @段首空格
    r'(.{350,370}) $': r'\1',         # @段尾空格
    r' {2,}': ' ',    # @连续空格 多个空格改为单个

#=== 格鲁吉亚语专项修复 ===#
    r'[―─−‒–]': '—',    # @错误破折号
    r' +([,;:?!)…“».])': r'\1',    # @标点前有空格
    r'([,;:?!“»)])([ა-ჰ])': r'\1 \2',    # @标点后面没有空格
    r'([ა-ჰ0-9])([(„«])': r'\1 \2',    # @前括号/前引号前面没有空格
    r'([(„«]) +': r'\1',    # @前括号/前引号后面有空格
#   r'\.{3}': '…',    # 三个句点改为省略号（根据需要启用）
#   直引号 "'‘’ 无法自动修复：需判断开闭方向（„ 还是 “），请人工处理
#   全角标点 ，。、：；！？（） 未自动替换：出现频率低且易与数字格式冲突，请人工处理
}
