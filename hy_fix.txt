# 自动修复规则文件 - 亚美尼亚语
# 使用说明:
# 1. 此文件放在 src/config/ 目录，命名格式：{语种代码}_fix.txt
# 2. # 开头为注释行，不生效
# 3. 格式为 Python dict：{r'正则匹配模式': '替换内容'}
# 4. 替换作用于 docx 表格外语列（最右侧列）的每个段落，^ $ 即段首段尾
# 5. 若无此文件，修复步骤自动跳过
# 6. 注释里的 @标签 需与 hy_txt.txt 中的检查描述完全一致，才能关联修复状态
# 7. 规则按书写顺序依次执行，隐形字符清理务必排在空格规则之前
#
{
#=== 隐形字符清理（须最先执行）===#
    r'[\u00ad\u200b-\u200f\u202a-\u202e\ufeff]': '',   # @不可见字符
    r'\u00a0': ' ',     # @不间断空格NBSP

#=== 空格修复 ===#
    r'^ +': '',    # @段首空格
    r' +$': '',    # @段尾空格
    r' {2,}': ' ',    # @连续空格 多个空格改为单个

#=== 亚美尼亚语专项修复 ===#
    r'[–─−]': '—',    # @错误破折号
    r'(?<!\d):(?![0-9/])': '։',    # @英文冒号应改为։ 章节数字之间的冒号保留
    r' +([-,.՞՜)»—~։])': r'\1',    # @标点前有空格
    r'([(«]) +': r'\1',    # @连字符/破折号/前括号/前引号后面有空格
    r'(?<=\S)([-—~]) +': r'\1',    # @连字符/破折号/前括号/前引号后面有空格 段首的对话破折号不动
    r'([)»։])([^-\s՞՜։,.»)])': r'\1 \2',    # @标点后面没有空格
    r',(?![0-9])([^-\s՞՜։,.»)])': r', \1',    # @逗号后面没有空格
    r'([^\s(«])([(«])': r'\1 \2',    # @前括号/前引号前面没有空格
#   以下规则默认关闭，按需启用：
#   r'\?': '՞',    # ՞ 应置于疑问词的最后一个元音之上，机械替换会留在句尾，位置错误
#   r'!': '՜',     # ՜ 同上
#   r'\.(?![0-9])([^-\s՞՜։,.»)])': r'. \1',    # 句点后补空格：会误伤 Ա.Բ. 类缩写
#   r'(\d),(\d{3})(?!\d)': r'\1.\2',    # 英文千位逗号改亚美尼亚千位点：与小数写法易混，建议只备注不自动改
#   直引号 "'“”„ 无法自动修复：需判断开闭方向（« 还是 »），请人工处理
#   全角标点 ，。、：；！？（） 未自动替换：与数字格式易冲突，请人工处理
}
